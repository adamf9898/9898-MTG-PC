<!-- Pack Size Control Widget -->
<div class="mtg-pack-ctrl" role="region" aria-label="Pack generator controls">
  <label for="pack-size">Pack size:</label>
  <input type="number" id="pack-size" min="1" max="30" value="15"
         aria-label="Number of cards in pack" />
  <button id="spawn-btn" onclick="spawnPack()">⚔ Spawn Pack</button>
  <button id="cancel-btn" onclick="cancelPack()" disabled>✕ Cancel</button>
  <div id="status" aria-live="polite" aria-atomic="true"></div>
  <div id="card-grid"></div>
</div>

<script>
  let _abort = null;

  async function spawnPack() {
    const size = Math.min(+document.getElementById('pack-size').value, 30);
    const status = document.getElementById('status');
    const grid   = document.getElementById('card-grid');
    const spawnBtn = document.getElementById('spawn-btn');
    const cancelBtn = document.getElementById('cancel-btn');

    grid.innerHTML = '';
    spawnBtn.disabled = true;
    cancelBtn.disabled = false;
    _abort = new AbortController();

    for (let i = 0; i < size; i++) {
      if (_abort.signal.aborted) break;
      status.textContent = `Fetching card ${i + 1} of ${size}…`;
      try {
        const res = await fetch('https://api.scryfall.com/cards/random',
          { signal: _abort.signal, headers: { 'Accept': 'application/json' } });
        const card = await res.json();
        if (card.image_uris) {
          const img = document.createElement('img');
          img.src = card.image_uris.normal || card.image_uris.small;
          img.alt = card.name;
          img.title = `${card.name} — ${card.type_line}`;
          grid.appendChild(img);
        }
        await new Promise(r => setTimeout(r, 350));
      } catch (e) {
        if (e.name !== 'AbortError') status.textContent = `Error: ${e.message}`;
        break;
      }
    }

    spawnBtn.disabled = false;
    cancelBtn.disabled = true;
    if (!_abort.signal.aborted) status.textContent = 'Pack complete!';
    _abort = null;
  }

  function cancelPack() {
    _abort?.abort();
    document.getElementById('status').textContent = 'Cancelled.';
    document.getElementById('spawn-btn').disabled = false;
    document.getElementById('cancel-btn').disabled = true;
  }
</script>
A self-contained widget. Copy-paste into any HTML page. Uses AbortController for cancellation and aria-live for screen-reader announcements during fetch.
