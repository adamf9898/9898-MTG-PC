JavaScript Loader Integration

Dynamic fetch patterns to load and display generator results from JavaScript

← Home Plugins Global Vars Blocks HTML Embed JS Loader Plugin UI Advanced
Perchance generators run on Perchance.org. The snippets below show how to use Scryfall API directly from JavaScript — the same data source the Lua script uses — to build a browser-side version of the pack generator.

Minimal Fetch — Single Random Card

JavaScript (ES2020+)
async function fetchRandomCard() {
  const res = await fetch('https://api.scryfall.com/cards/random', {
    headers: {
      'Accept': 'application/json',
      'User-Agent': 'MTG-Manual-Demo/1.0'
    }
  });
  if (!res.ok) throw new Error(`Scryfall ${res.status}`);
  const card = await res.json();
  return {
    name:     card.name,
    imageUrl: card.image_uris?.normal ?? card.image_uris?.small,
    cmc:      card.cmc,
    typeLine: card.type_line
  };
}

// Usage
fetchRandomCard().then(card => {
  document.getElementById('card-name').textContent = card.name;
  document.getElementById('card-img').src = card.imageUrl;
});
Fetches one random card from Scryfall. The ?? operator falls back to small when normal is absent (e.g., tokens). Mirrors the fetchCard() and onCardFetched() Lua functions.

Pack Generator — 15 Cards with Rate Limiting

JavaScript — Pack generator
const MIN_DELAY_MS = 350; // match Lua MIN_DELAY

function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function fetchCard(retries = 3) {
  for (let attempt = 0; attempt < retries; attempt++) {
    try {
      const res = await fetch('https://api.scryfall.com/cards/random', {
        headers: { 'Accept': 'application/json', 'User-Agent': 'MTG-Pack-JS/1.0' }
      });
      if (res.status === 429) {
        await sleep(500 * Math.pow(2, attempt)); // exponential backoff
        continue;
      }
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const data = await res.json();
      if (!data.image_uris) continue; // skip tokens
      return data;
    } catch (e) {
      if (attempt === retries - 1) return null;
    }
  }
  return null;
}

async function spawnPack(size = 15, onCard = () => {}) {
  const cards = [];
  for (let i = 0; i < size; i++) {
    const card = await fetchCard();
    if (card) { cards.push(card); onCard(card, i); }
    if (i < size - 1) await sleep(MIN_DELAY_MS);
  }
  return cards;
}

// Usage
const container = document.getElementById('pack-container');
spawnPack(15, (card) => {
  const img = document.createElement('img');
  img.src = card.image_uris.normal;
  img.alt = card.name;
  img.title = `${card.name} — ${card.type_line}`;
  img.style.cssText = 'width:120px;border-radius:6px;margin:4px;';
  container.appendChild(img);
});
A full browser-side port of the Lua spawn logic: exponential backoff, rate limiting, and token skipping. The onCard callback lets you update the UI incrementally as each card arrives.

Abort Controller — Cancellable Spawn

JavaScript — Cancellable fetch
let abortController = null;

async function startPack() {
  abortController = new AbortController();
  const { signal } = abortController;
  try {
    for (let i = 0; i < 15; i++) {
      if (signal.aborted) break;
      const res = await fetch('https://api.scryfall.com/cards/random', { signal });
      const card = await res.json();
      renderCard(card);
      await sleep(350);
    }
  } catch (e) {
    if (e.name !== 'AbortError') console.error(e);
  }
}

function cancelPack() {
  abortController?.abort();
}
Uses the AbortController API to cancel an in-progress pack fetch — equivalent to TTS's _isSpawning guard but with immediate abort capability.
