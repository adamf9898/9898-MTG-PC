usePack — Custom Hook

React — usePack.js
import { useState, useCallback, useRef } from 'react';

const SCRY_RANDOM = 'https://api.scryfall.com/cards/random';

export function usePack() {
  const [cards, setCards]       = useState([]);
  const [status, setStatus]     = useState('idle'); // idle | loading | done | error
  const [message, setMessage]   = useState('');
  const abortRef                = useRef(null);

  const spawnPack = useCallback(async (size = 15) => {
    abortRef.current?.abort();
    abortRef.current = new AbortController();
    setCards([]);
    setStatus('loading');

    for (let i = 0; i < size; i++) {
      if (abortRef.current.signal.aborted) break;
      setMessage(`Fetching ${i + 1} / ${size}`);
      try {
        const res  = await fetch(SCRY_RANDOM, { signal: abortRef.current.signal });
        const data = await res.json();
        if (data.image_uris) {
          setCards(prev => [...prev, data]);
        }
        await new Promise(r => setTimeout(r, 350));
      } catch (e) {
        if (e.name !== 'AbortError') { setStatus('error'); return; }
        break;
      }
    }
    setStatus('done');
    setMessage('');
  }, []);

  const cancel = useCallback(() => {
    abortRef.current?.abort();
    setStatus('idle');
  }, []);

  return { cards, status, message, spawnPack, cancel };
}
A reusable hook that encapsulates fetch, abort, and state. Import it into any component.

PackGrid Component

React — PackGrid.jsx
import { usePack } from './usePack';

export default function PackGrid() {
  const { cards, status, message, spawnPack, cancel } = usePack();

  return (
    <div style={{ background:'#0a0a0a', border:'2px solid #cc0000',
                  borderRadius:10, padding:'1.2rem', fontFamily:'monospace' }}>
      <div style={{ display:'flex', gap:8, marginBottom:'0.8rem', alignItems:'center' }}>
        <input type="number" defaultValue={15} min={1} max={30} id="react-size"
               style={{ width:60, background:'#1a1a1a', border:'1px solid #2a0000',
                        color:'#e8e8e8', padding:'4px', borderRadius:4 }} />
        <button onClick={() => spawnPack(+document.getElementById('react-size').value)}
                disabled={status === 'loading'}
                style={{ background:'#cc0000', color:'#fff', border:'none',
                         padding:'6px 14px', borderRadius:4, cursor:'pointer' }}>
          {status === 'loading' ? 'Loading…' : '⚔ Spawn Pack'}
        </button>
        {status === 'loading' &&
          <button onClick={cancel}
                  style={{ background:'transparent', border:'1px solid #cc0000',
                           color:'#cc0000', padding:'6px 14px', borderRadius:4, cursor:'pointer' }}>
            Cancel
          </button>
        }
      </div>
      {message && <p style={{ color:'#888', fontSize:'.8rem', margin:'0 0 .6rem' }}>{message}</p>}
      <div style={{ display:'flex', flexWrap:'wrap', gap:6 }}>
        {cards.map((c, i) => (
          <img key={i} src={c.image_uris.normal} alt={c.name}
               title={`${c.name} — ${c.type_line}`}
               style={{ width:100, borderRadius:6 }} />
        ))}
      </div>
    </div>
  );
}

3D Card Fan Scene

Three.js — card-fan.js
import * as THREE from 'three';

const SCRY_RANDOM = 'https://api.scryfall.com/cards/random';
const CARD_W = 1.4, CARD_H = 2.0, CARD_D = 0.01;

async function fetchCardImage() {
  const res  = await fetch(SCRY_RANDOM);
  const data = await res.json();
  return data.image_uris?.normal ?? null;
}

export async function createCardFanScene(canvas, packSize = 5) {
  const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true });
  renderer.setSize(canvas.clientWidth, canvas.clientHeight);
  renderer.setClearColor(0x0a0a0a);

  const scene  = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(40, canvas.clientWidth / canvas.clientHeight, 0.1, 100);
  camera.position.set(0, 0, 6);

  scene.add(new THREE.AmbientLight(0xffffff, 0.8));
  const dir = new THREE.DirectionalLight(0xffffff, 0.6);
  dir.position.set(2, 4, 4);
  scene.add(dir);

  const loader    = new THREE.TextureLoader();
  const backTex   = loader.load('https://i.imgur.com/yHEAQ88.jpeg');
  const geo       = new THREE.BoxGeometry(CARD_W, CARD_H, CARD_D);
  const cards     = [];

  for (let i = 0; i < packSize; i++) {
    const url    = await fetchCardImage();
    const faceTex = url ? await new Promise(r => loader.load(url, r)) : backTex;
    const mats   = [
      new THREE.MeshStandardMaterial({ color: 0x111111 }),
      new THREE.MeshStandardMaterial({ color: 0x111111 }),
      new THREE.MeshStandardMaterial({ color: 0x111111 }),
      new THREE.MeshStandardMaterial({ color: 0x111111 }),
      new THREE.MeshStandardMaterial({ map: faceTex }),
      new THREE.MeshStandardMaterial({ map: backTex  }),
    ];
    const mesh = new THREE.Mesh(geo, mats);
    const angle = (i / (packSize - 1) - 0.5) * Math.PI * 0.5;
    mesh.position.set(Math.sin(angle) * 2.5, 0, Math.cos(angle) * -0.5);
    mesh.rotation.y = -angle;
    scene.add(mesh);
    cards.push(mesh);
    await new Promise(r => setTimeout(r, 350));
  }

  let t = 0;
  function animate() {
    requestAnimationFrame(animate);
    t += 0.005;
    cards.forEach((c, i) => { c.position.y = Math.sin(t + i * 0.5) * 0.08; });
    renderer.render(scene, camera);
  }
  animate();
}
Creates a floating fan of card meshes in a Three.js scene. Each card fetches a real Scryfall image as its face texture. Cards bob gently on a sine wave.
Requires Phaser 3.60+. Install: npm install phaser

Pack Scene

Phaser 3 — PackScene.js
import Phaser from 'phaser';

const SCRY_RANDOM = 'https://api.scryfall.com/cards/random';

export class PackScene extends Phaser.Scene {
  constructor() { super({ key: 'PackScene' }); }

  async create() {
    this.cards = [];
    this.statusText = this.add.text(10, 10, 'Loading pack…', {
      fontFamily: 'monospace', fontSize: '14px', color: '#cc0000'
    });

    for (let i = 0; i < 15; i++) {
      this.statusText.setText(`Fetching card ${i + 1} / 15`);
      try {
        const res  = await fetch(SCRY_RANDOM);
        const data = await res.json();
        if (data.image_uris?.normal) {
          const key = `card_${i}`;
          await this.loadImageAsync(key, data.image_uris.normal);
          const img = this.add.image(80 + i * 60, 200, key)
            .setScale(0.18)
            .setInteractive({ cursor: 'pointer' });
          img.on('pointerover', () => img.setScale(0.26));
          img.on('pointerout',  () => img.setScale(0.18));
          img.setData('card', data);
          this.cards.push(img);
        }
        await new Promise(r => setTimeout(r, 350));
      } catch (e) { console.warn('Card fetch failed', e); }
    }
    this.statusText.setText('Pack ready!');
  }

  loadImageAsync(key, url) {
    return new Promise((resolve, reject) => {
      if (this.textures.exists(key)) { resolve(); return; }
      this.load.image(key, url);
      this.load.once('complete', resolve);
      this.load.once('loaderror', reject);
      this.load.start();
    });
  }
}
A Phaser Scene that asynchronously fetches and displays 15 cards. Cards are interactive — hovering scales them up for inspection. Integrates directly with the existing 9898-MTG-Chaos-RPG-2026 Phaser setup.
